Building a Simple Chatbot with OpenAI

This code demonstrates how to send a prompt and receive a response.


import os
import requests
import json

# NEVER hardcode your API key! Always use environment variables.
API_KEY = os.getenv("OPENAI_API_KEY")
URL = "https://api.openai.com/v1/chat/completions"

# 1. Define the Headers (Authentication)
headers = {
    "Authorization": f"Bearer {API_KEY}",
    "Content-Type": "application/json"
}

# 2. Define the Payload (The Data)
payload = {
    "model": "gpt-3.5-turbo",
    "messages": [
        {"role": "system", "content": "You are a helpful coding assistant."},
        {"role": "user", "content": "Explain what a REST API is in one sentence."}
    ],
    "temperature": 0.7 # Creativity control (0.0 to 1.0)
}

# 3. Make the API Call
response = requests.post(URL, headers=headers, json=payload)

# 4. Handle the JSON Response
if response.status_code == 200:
    data = response.json()
    # Extract the AI's reply from the JSON structure
    ai_reply = data["choices"][0]["message"]["content"]
    print(f"AI: {ai_reply}")
else:
    print(f"Error: {response.status_code} - {response.text}")
