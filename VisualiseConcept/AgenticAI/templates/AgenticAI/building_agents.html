Code Example: Basic LangChain Agent with a Math Tool


from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.tools import Tool

def calculate_length(text):
    """Returns the length of a string."""
    return len(text)

# 1. Define Tools
tools = [
    Tool(
        name="StringLengthCalculator",
        func=calculate_length,
        description="Useful for when you need to count characters in a word or phrase."
    )
]

# 2. Init LLM
llm = ChatOpenAI(temperature=0)

# 3. Initialize Agent
agent = initialize_agent(
    tools, 
    llm, 
    agent=AgentType.ZERO_SHOT_REACT_DESCRIPTION, 
    verbose=True
)

# 4. Run Agent
agent.run("How many characters are in the word 'Supercalifragilisticexpialidocious'?")
