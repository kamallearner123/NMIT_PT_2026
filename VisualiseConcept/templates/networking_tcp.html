4. A Simple TCP Echo Client (C)

This code establishes a TCP connection to a server and sends a simple "Hello" message.

#include <stdio.h>       // Input/Output functions
#include <sys/socket.h>  // Main socket definitions
#include <arpa/inet.h>   // Functions for internet operations (IP conversion)
#include <unistd.h>      // Standard symbolic constants and types
#include <string.h>      // String handling functions

int main() {
    int sock = 0;                        // File descriptor for our socket
    struct sockaddr_in serv_addr;         // Structure for server address info
    char *hello = "Hello from Client";   // Message to send
    char buffer[1024] = {0};             // Buffer to store server response

    // 1. Create the socket (AF_INET = IPv4, SOCK_STREAM = TCP)
    if ((sock = socket(AF_INET, SOCK_STREAM, 0)) < 0) {
        printf("\n Socket creation error \n");
        return -1;
    }

    // 2. Configure server address
    serv_addr.sin_family = AF_INET;      // Set address family to IPv4
    serv_addr.sin_port = htons(80);      // Set port to 80 (HTTP)

    // 3. Convert IP address from text to binary form
    // We use a dummy IP for local demonstration
    if(inet_pton(AF_INET, "127.0.0.1", &serv_addr.sin_addr) <= 0) {
        printf("\nInvalid address/ Address not supported \n");
        return -1;
    }

    // 4. Connect to the server (This triggers the 3-Way Handshake)
    // Note: This will fail in this lab since no server is listening on port 80
    if (connect(sock, (struct sockaddr *)&serv_addr, sizeof(serv_addr)) < 0) {
        printf("\nConnection Failed (Expected as no server is listening) \n");
        return -1;
    }

    // 5. Send data to the server
    send(sock, hello, strlen(hello), 0);
    printf("Hello message sent\n");

    return 0;
}
