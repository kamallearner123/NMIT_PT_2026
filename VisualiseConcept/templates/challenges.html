Traffic Congestion Analyzer

Pattern: Sliding Window Domain: Smart City

Scenario: A traffic control system identifies the most congested 5-minute window to redirect vehicles.

def max_traffic_window(data, k):
    max_sum = curr_sum = sum(data[:k])
    for i in range(k, len(data)):
        curr_sum += data[i] - data[i - k]
        max_sum = max(max_sum, curr_sum)
    return max_sum

Autonomous Convoy Synchronizer

Pattern: Two Pointers Domain: Automotive

Scenario: Self-driving trucks pair up if their speeds match a strict differential delta (diff = 5).

def find_convoy_pairs(speeds, target_diff):
    left, right = 0, 1
    pairs = []
    while left < len(speeds) and right < len(speeds):
        diff = speeds[right] - speeds[left]
        if left != right and diff == target_diff:
            pairs.append((speeds[left], speeds[right]))
            left += 1; right += 1
        elif diff < target_diff: right += 1
        else: left += 1
    return pairs

def has_routing_loop(head):
    slow, fast = head, head
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
        if slow == fast:
            return True # Loop Detected
    return False

Hospital ER Triage

Pattern: Priority Queue (Heap) Domain: Healthcare

Scenario: An Emergency Room receives patients with varying severity. Treat the most critical patient next.

import heapq
def triage_patients(patients):
    queue = []
    # Max-Heap using negative severity
    for name, sev in patients:
        heapq.heappush(queue, (-sev, name))
        
    return [heapq.heappop(queue)[1] for _ in range(len(queue))]

Stock Market Peak Finder

Pattern: Monotonic Stack Domain: Finance

Scenario: A trading bot needs to know the next profitable sell opportunity for every historical data point.

def next_greater_price(prices):
    result = [-1] * len(prices)
    stack = []
    for i, p in enumerate(prices):
        while stack and prices[stack[-1]] < p:
            result[stack.pop()] = p
        stack.append(i)
    return result
